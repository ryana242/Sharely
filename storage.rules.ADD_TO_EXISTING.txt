    function booking(bookingId) {
      return firestore.get(/databases/(default)/documents/bookings/$(bookingId)).data;
    }
    function participant(bookingId) {
      return request.auth != null
        && (booking(bookingId).borrowerId == request.auth.uid
          || firestore.get(
            /databases/(default)/documents/items/$(booking(bookingId).itemId))
            .data.ownerId == request.auth.uid);
    }
    match /transactionEvidence/{bookingId}/{userId}/{photoId} {
      allow read: if participant(bookingId);
      allow write: if request.auth != null
        && request.auth.uid == userId
        && participant(bookingId)
        && resource == null
        && request.resource.size < 10 * 1024 * 1024
        && request.resource.contentType.matches('image/.*');
    }
